        // 履歴ベースで重複を避けつつランダムに選ぶ関数
        function getVariationText(optionKey) {
            const texts = reviewVariations[optionKey];
            if (!texts) return "";

            const historyKey = 'review_history_' + optionKey;
            let history = [];
            try {
                history = JSON.parse(localStorage.getItem(historyKey)) || [];
            } catch (e) {
                history = [];
            }

            // 直近（最大5回分）使われていないインデックスの候補を作成
            const maxHistory = Math.min(5, texts.length - 1);
            const recentUsed = history.slice(-maxHistory);
            let availableIndices = [];
            
            for (let i = 0; i < texts.length; i++) {
                if (!recentUsed.includes(i)) {
                    availableIndices.push(i);
                }
            }

            // 万が一候補が空なら全インデックスを候補にする
            if (availableIndices.length === 0) {
                availableIndices = texts.map((_, i) => i);
            }

            // 候補の中から完全にランダムで1つ選択（毎回違うパターンになる）
            const chosenIndex = availableIndices[Math.floor(Math.random() * availableIndices.length)];

            // 履歴を更新して保存
            history.push(chosenIndex);
            if (history.length > 10) history.shift();
            localStorage.setItem(historyKey, JSON.stringify(history));

            return texts[chosenIndex];
        }

        // ランダムな自然接続フレーズ（冒頭や段落のマンネリ防止用）
        function getRandomConnector(type) {
            const connectors = {
                symptomIntro: ["", "", "以前から", "実は", "ここ最近、"],
                praiseIntro: ["", "", "特に印象的だったのは、", "実際に通ってみて感じたのは、", "何より嬉しいのが、"],
                accessIntro: ["", "", "また、", "通いやすさの面でも、", "それに加えて、"],
                resultIntro: ["", "", "おかげさまで、", "結果として、", "今では"]
            };
            const list = connectors[type] || [""];
            return list[Math.floor(Math.random() * list.length)];
        }

        function generateReview() {
            const q1Key = document.getElementById('q1').value;
            const q2Key = document.getElementById('q2').value;
            const q3Key = document.getElementById('q3').value;
            const q4Key = document.getElementById('q4').value;
            const q5Key = document.getElementById('q5').value;
            
            const errorMsg = document.getElementById('errorMessage');
            const resultArea = document.getElementById('resultArea');

            // いずれかが未選択の場合、エラーメッセージを表示して処理を中断
            if (!q1Key || !q2Key || !q3Key || !q4Key || !q5Key) {
                errorMsg.classList.remove('hidden');
                // 少しスクロールしてエラーを見やすくする
                window.scrollBy({ top: -50, behavior: 'smooth' });
                return;
            }

            // エラーがない場合はメッセージを隠す
            errorMsg.classList.add('hidden');

            // 各キーからバリエーション文章を取得
            const text1 = getVariationText(q1Key);
            const text2 = getVariationText(q2Key);
            const text3 = getVariationText(q3Key);
            const text4 = getVariationText(q4Key);
            const text5 = getVariationText(q5Key);

            // 文章の構成（順番）も毎回変わるように6パターンを用意し、履歴ベースでランダム選択
            const structureHistoryKey = 'review_structure_history';
            let structHistory = [];
            try {
                structHistory = JSON.parse(localStorage.getItem(structureHistoryKey)) || [];
            } catch (e) {
                structHistory = [];
            }

            const totalStructures = 6;
            const recentStructs = structHistory.slice(-4); // 直近4回と同じ構成を避ける
            let availableStructs = [];
            for (let i = 0; i < totalStructures; i++) {
                if (!recentStructs.includes(i)) {
                    availableStructs.push(i);
                }
            }
            if (availableStructs.length === 0) {
                availableStructs = [0, 1, 2, 3, 4, 5];
            }

            const structureIndex = availableStructs[Math.floor(Math.random() * availableStructs.length)];
            structHistory.push(structureIndex);
            if (structHistory.length > 10) structHistory.shift();
            localStorage.setItem(structureHistoryKey, JSON.stringify(structHistory));

            let reviewText = "";
            switch (structureIndex) {
                case 0:
                    // パターンA：王道・時系列型（きっかけ → 施術・雰囲気・通いやすさ → 変化）
                    reviewText = `${text1}\n\n${getRandomConnector('praiseIntro')}${text3}\n${text2}\n${getRandomConnector('accessIntro')}${text4}\n\n${getRandomConnector('resultIntro')}${text5}`;
                    break;
                case 1:
                    // パターンB：結論・感動ファースト型（変化・おすすめ → きっかけ・施術 → 雰囲気・通いやすさ）
                    reviewText = `${text5}\n\n${getRandomConnector('symptomIntro')}${text1}\n${text3}\n\n${text2}\n${getRandomConnector('accessIntro')}${text4}`;
                    break;
                case 2:
                    // パターンC：立地・雰囲気から入る自然な導入型（通いやすさ・雰囲気 → きっかけ・施術 → 変化）
                    reviewText = `${text4}\n${text2}\n\n${getRandomConnector('symptomIntro')}${text1}\n${getRandomConnector('praiseIntro')}${text3}\n\n${getRandomConnector('resultIntro')}${text5}`;
                    break;
                case 3:
                    // パターンD：施術の良さピックアップ型（施術 → きっかけ → 変化 → 通いやすさ・雰囲気）
                    reviewText = `${text3}\n${text1}\n\n${getRandomConnector('resultIntro')}${text5}\n\n${getRandomConnector('accessIntro')}${text4}\n${text2}`;
                    break;
                case 4:
                    // パターンE：院内の雰囲気ファースト型（雰囲気 → きっかけ・施術 → 変化・通いやすさ）
                    reviewText = `${text2}\n\n${getRandomConnector('symptomIntro')}${text1}\n${text3}\n\n${getRandomConnector('resultIntro')}${text5}\n${text4}`;
                    break;
                case 5:
                    // パターンF：お悩み解決ストーリー型（きっかけ・変化 → 施術・通いやすさ・雰囲気）
                    reviewText = `${text1}\n${getRandomConnector('resultIntro')}${text5}\n\n${getRandomConnector('praiseIntro')}${text3}\n${text4}\n${text2}`;
                    break;
            }
            
            document.getElementById('reviewText').value = reviewText;
            
            // 結果エリアを表示
            resultArea.classList.remove('hidden');
            document.getElementById('googleBtn').classList.add('hidden'); // コピーされるまで隠す
            document.getElementById('copyToast').classList.add('hidden');
            
            // 結果エリアへスムーズにスクロール
            setTimeout(() => {
                resultArea.scrollIntoView({ behavior: 'smooth', block: 'start' });
            }, 100);
        }

        function copyText() {
            const textarea = document.getElementById('reviewText');
            // テキストエリアを選択状態にする（ユーザーへの視覚的フィードバック）
            textarea.select();
            textarea.setSelectionRange(0, 99999); // モバイルブラウザ対応

            let copied = false;
            try {
                // iFrame内や幅広い環境で動作しやすい execCommand を優先実行
                copied = document.execCommand('copy');
            } catch (e) {
                copied = false;
            }

            if (copied) {
                document.getElementById('copyToast').classList.remove('hidden');
                document.getElementById('googleBtn').classList.remove('hidden');
                document.getElementById('googleBtn').classList.add('flex');
            } else if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(textarea.value).then(() => {
                    document.getElementById('copyToast').classList.remove('hidden');
                    document.getElementById('googleBtn').classList.remove('hidden');
                    document.getElementById('googleBtn').classList.add('flex');
                }).catch(() => {
                    showCopyError(textarea);
                });
            } else {
                showCopyError(textarea);
            }
        }

        function showCopyError(textarea) {
            if (!document.getElementById('copyErrorMsg')) {
                const errorDiv = document.createElement('div');
                errorDiv.id = 'copyErrorMsg';
                errorDiv.className = "text-red-500 text-sm mt-2 mb-2 font-bold text-center";
                errorDiv.innerText = "自動コピーできませんでした。文章を長押しして手動でコピーしてください。";
                textarea.parentNode.insertBefore(errorDiv, textarea.nextSibling);
            }
        }

        function copyText() {
            const textarea = document.getElementById('reviewText');
            // テキストエリアを選択状態にする（ユーザーへの視覚的フィードバック）
            textarea.select();
            textarea.setSelectionRange(0, 99999); // モバイルブラウザ対応

            // クリップボードAPIを使用してコピー
            navigator.clipboard.writeText(textarea.value).then(() => {
                // コピー成功時の処理
                document.getElementById('copyToast').classList.remove('hidden');
                document.getElementById('googleBtn').classList.remove('hidden');
                document.getElementById('googleBtn').classList.add('flex'); // hiddenを外した後にflexを当てる
            }).catch(err => {
                // 古いブラウザ等で失敗した場合のフォールバック
                try {
                    document.execCommand('copy');
                    document.getElementById('copyToast').classList.remove('hidden');
                    document.getElementById('googleBtn').classList.remove('hidden');
                    document.getElementById('googleBtn').classList.add('flex');
                } catch (e) {
                    // フォールバックも失敗した場合
                    const errorDiv = document.createElement('div');
                    errorDiv.className = "text-red-500 text-sm mt-2 font-bold";
                    errorDiv.innerText = "自動コピーできませんでした。長押しして手動でコピーしてください。";
                    textarea.parentNode.insertBefore(errorDiv, textarea.nextSibling);
                }
            });
        }

        function showQRCode() {
            const modal = document.getElementById('qrModal');
            const container = document.getElementById('qrImageContainer');
            modal.classList.remove('hidden');
            
            // 現在のページのURLを取得
            let currentUrl = window.location.href;
            
            // プレビュー環境など、特殊なURLの場合はダミーURLを使用（エラー回避）
            if(currentUrl.startsWith('about:') || currentUrl.startsWith('blob:') || currentUrl.startsWith('file:')) {
                currentUrl = 'https://example.com/saiin-seikotsuin-review'; 
            }

            // QRコード生成APIを呼び出して画像タグを作成
            const qrApiUrl = `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${encodeURIComponent(currentUrl)}&color=ea580c`;
            
            container.innerHTML = `<img src="${qrApiUrl}" alt="QR Code" class="w-48 h-48 mx-auto shadow-sm rounded-xl">`;
        }
